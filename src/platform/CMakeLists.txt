target_include_directories(libcdi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(WIN32)
  target_include_directories(libcdi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_include_directories(libcdi PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/windows)
  target_sources(libcdi PUBLIC windows/cdi_pread.c)
elseif(UNIX)
  target_include_directories(libcdi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_include_directories(libcdi PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/unix)
  target_sources(libcdi PUBLIC unix/cdi_pread.c)
  target_link_libraries(libcdi PUBLIC m)
endif()
