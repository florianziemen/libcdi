find_package(Ruby REQUIRED)

set(FORTRAN_INTERFACE_SOURCE ${CMAKE_SOURCE_DIR}/interfaces/f2003)
set(FORTRAN_INTERFACE_BINARY_DIR ${CMAKE_BINARY_DIR}/interfaces/f2003)
set(GEN_FILE ${FORTRAN_INTERFACE_BINARY_DIR}/mo_cdi.f90)
set(CDI_INTERFACE_FILE ${CMAKE_SOURCE_DIR}/src/cdi.h)

add_custom_command(
  OUTPUT ${GEN_FILE}
  COMMAND ${RUBY_EXECUTABLE}
  ${FORTRAN_INTERFACE_SOURCE}/bindGen.rb
  ${CDI_INTERFACE_FILE}
  ${GEN_FILE}
  DEPENDS
  ${CDI_INTERFACE_FILE}
  ${FORTRAN_INTERFACE_SOURCE}/bindGen.rb
  WORKING_DIRECTORY ${FORTRAN_INTERFACE_BINARY_DIR}
  COMMENT "Generating Fortran source mo_cdi.f90"
  VERBATIM
)

add_custom_target(generate_modules ALL
    DEPENDS ${GEN_FILE}
)
